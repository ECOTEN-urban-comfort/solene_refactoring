435static inline int
436cs_math_binom(int n,
437 int k)
438{
439 int ret = 1;
440 assert(n >= k);
441
442 const int n_iter = (k > n-k) ? n-k : k;
443 for (int j = 1; j <= n_iter; j++, n--) {
444 if (n % j == 0)
445 ret *= n/j;
446 else if (ret % j == 0)
447 ret = ret/j*n;
448 else
449 ret = (ret*n)/j;
450 }
451
452 return ret;
453}
