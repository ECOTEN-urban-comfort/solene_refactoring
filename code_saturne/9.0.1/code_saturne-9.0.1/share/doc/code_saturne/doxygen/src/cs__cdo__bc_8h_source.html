101/* Structure specific to store data related to the definition of boundary
102 * conditions on boundary faces.
103 *
104 * For of scalar-valued equations, only some the classical (Dirichlet, Neumann
105 * and Robin types are available. Other types of boundary conditions are
106 * possible for vector-valued equations.
107 */
108
109typedef struct {
110
111 bool is_steady; /* Do we need to update BC faces during the
112 computation */
113 cs_lnum_t n_b_faces; /* Number of boundary faces */
114
115 /* Type of boundary conditions associated to a face. Size: n_b_faces */
116
117 cs_flag_t *flag;
118
119 /* Id of the boundary condition definition or CS_BC_DEFAULT (=-1) if this face
120 is related to the default boundary condition. Size = n_b_faces */
121
122 short int *def_ids;
123
124 /* List of face ids by type of boundary conditions. Homogeneous types don't
125 * need to rely on a definition since it can be the default bc. Moreover, some
126 * optimizations can be performed that's why they are stored separately
127 */
128
129 /* Dirichlet */
130
131 cs_lnum_t n_hmg_dir_faces;
132 cs_lnum_t *hmg_dir_ids;
133 cs_lnum_t n_nhmg_dir_faces;
134 cs_lnum_t *nhmg_dir_ids;
135
136 /* Neumann */
137
138 cs_lnum_t n_hmg_neu_faces;
139 cs_lnum_t *hmg_neu_ids;
140 cs_lnum_t n_nhmg_neu_faces;
141 cs_lnum_t *nhmg_neu_ids;
142
143 /* Robin */
144
145 cs_lnum_t n_robin_faces;
146 cs_lnum_t *robin_ids;
147
148 /* Sliding wall */
149
150 cs_lnum_t n_sliding_faces;
151 cs_lnum_t *sliding_ids;
152
153 /* Circulation */
154
155 cs_lnum_t n_circulation_faces;
156 cs_lnum_t *circulation_ids;
157
158} cs_cdo_bc_face_t;

177static inline void
178cs_cdo_bc_get_desc(cs_flag_t bc_flag,
179 char *desc)
180{
181 if (desc == NULL)
182 bft_error(__FILE__, __LINE__, 0,
183 " %s: Empty desciption buffer.", __func__);
184
185 switch (bc_flag) {
186
187 case CS_CDO_BC_HMG_DIRICHLET:
188 sprintf(desc, "%s", "Homogenous Dirichlet");
189 break;
190 case CS_CDO_BC_DIRICHLET:
191 sprintf(desc, "%s", "Dirichlet");
192 break;
193 case CS_CDO_BC_SYMMETRY:
194 sprintf(desc, "%s", "Sliding for vectors");
195 break;
196 case CS_CDO_BC_HMG_NEUMANN:
197 sprintf(desc, "%s", "Homogeneous Neumann");
198 break;
199 case CS_CDO_BC_NEUMANN:
200 sprintf(desc, "%s", "Neumann");
201 break;
202 case CS_CDO_BC_FULL_NEUMANN:
203 sprintf(desc, "%s", "Full Neumann");
204 break;
205 case CS_CDO_BC_ROBIN:
206 sprintf(desc, "%s", "Robin");
207 break;
208 case CS_CDO_BC_TANGENTIAL_DIRICHLET:
209 sprintf(desc, "%s", "Dirichlet on the tangential component");
210 break;
211
212 default:
213 bft_error(__FILE__, __LINE__, 0,
214 "%s: Invalid case. Please contact the support.\n", __func__);
215 break;
216 }
217}

230static inline cs_flag_t
231cs_cdo_bc_get_flag(cs_param_bc_type_t bc_type)
232{
233 cs_flag_t ret_flag;
234
235 switch (bc_type) {
236
237 case CS_BC_HMG_DIRICHLET:
238 ret_flag = CS_CDO_BC_HMG_DIRICHLET;
239 break;
240 case CS_BC_DIRICHLET:
241 ret_flag = CS_CDO_BC_DIRICHLET;
242 break;
243 case CS_BC_SYMMETRY:
244 ret_flag = CS_CDO_BC_SYMMETRY;
245 break;
246 case CS_BC_NEUMANN:
247 ret_flag = CS_CDO_BC_NEUMANN;
248 break;
249 case CS_BC_NEUMANN_FULL:
250 ret_flag = CS_CDO_BC_FULL_NEUMANN;
251 break;
252 case CS_BC_HMG_NEUMANN:
253 ret_flag = CS_CDO_BC_HMG_NEUMANN;
254 break;
255 case CS_BC_GENERALIZED_SYM:
256 bft_error(__FILE__, __LINE__, 0,
257 _("invalid boundary condition CS_BC_GENERALIZED_SYM for CDO"));
258 ret_flag = 0;
259 break;
260 case CS_BC_ROBIN:
261 ret_flag = CS_CDO_BC_ROBIN;
262 break;
263 case CS_BC_CIRCULATION:
264 ret_flag = CS_CDO_BC_TANGENTIAL_DIRICHLET;
265 break;
266 case CS_BC_WALL_MODELLED:
267 ret_flag = CS_CDO_BC_WALL_PRESCRIBED; /* TO BE CHECKED */
268 break;
269
270 default:
271 ret_flag = 0; /* Not handle automatically */
272 break;
273 }
274
275 return ret_flag;
276}
