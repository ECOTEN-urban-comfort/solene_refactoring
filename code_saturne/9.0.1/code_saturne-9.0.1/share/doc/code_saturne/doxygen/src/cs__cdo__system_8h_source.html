444static inline cs_cdo_system_matrix_class_t
445cs_cdo_system_which_matrix_class(cs_param_solver_class_t solver_class)
446{
447 switch (solver_class) {
448
449 case CS_PARAM_SOLVER_CLASS_CS:
450 return CS_CDO_SYSTEM_MATRIX_CS;
451 break;
452
453 case CS_PARAM_SOLVER_CLASS_HYPRE:
454#if defined(HAVE_HYPRE)
455 return CS_CDO_SYSTEM_MATRIX_HYPRE;
456#else
457 return CS_CDO_SYSTEM_MATRIX_CS;
458#endif
459 break;
460
461 default:
462 return CS_CDO_SYSTEM_MATRIX_CS;
463 break;
464
465 }
466}

482static inline cs_cdo_system_matrix_class_t
483cs_cdo_system_get_matrix_class(const cs_cdo_system_helper_t *sh,
484 int block_id)
485{
486 if (sh == NULL)
487 return CS_CDO_SYSTEM_MATRIX_NONE;
488 if (block_id < 0 || block_id >= sh->n_blocks)
489 return CS_CDO_SYSTEM_MATRIX_NONE;
490
491 cs_cdo_system_block_t *b = sh->blocks[block_id];
492
493 assert(b != NULL);
494 return b->info.matrix_class;
495}
