346
347static inline cs_field_t *
348cs_advection_field_get_field(const cs_adv_field_t *adv,
349 cs_mesh_location_type_t ml_type)
350{
351 cs_field_t *f = NULL;
352 if (adv == NULL)
353 return f;
354
355 switch (ml_type) {
356
357 case CS_MESH_LOCATION_CELLS:
358 if (adv->cell_field_id > -1)
359 f = cs_field_by_id(adv->cell_field_id);
360 else
361 f = NULL;
362 break;
363
364 case CS_MESH_LOCATION_INTERIOR_FACES:
365 if (adv->int_field_id > -1)
366 f = cs_field_by_id(adv->int_field_id);
367 else
368 f = NULL;
369 break;
370
371 case CS_MESH_LOCATION_BOUNDARY_FACES:
372 if (adv->bdy_field_id > -1)
373 f = cs_field_by_id(adv->bdy_field_id);
374 else
375 f = NULL;
376 break;
377
378 case CS_MESH_LOCATION_VERTICES:
379 if (adv->vtx_field_id > -1)
380 f = cs_field_by_id(adv->vtx_field_id);
381 else
382 f = NULL;
383 break;
384
385 default:
386 bft_error(__FILE__, __LINE__, 0,
387 " %s: Invalid mesh location type %d.\n"
388 " Stop retrieving the advection field.\n",
389 __func__, ml_type);
390 }
391
392 return f;
393}
